#!/bin/bash
# Cross-ACCOUNT satellite bootstrap (pre-kubelet ordering).
#
# A cross-account node must have the eks-hybrid:/// providerID in place BEFORE kubelet
# first registers — unlike same-account/cross-region, it does not get CCM's grace window.
# So we patch from a kubelet ExecStartPre drop-in: nodeadm-config writes the config files,
# then kubelet's ExecStartPre runs `xrn-install patch` (which sets providerID, hostname,
# labels, and installs the AssumeRole credential helper), and only then does kubelet start.
#
# cloud-boothook runs on first boot before cloud-init processes the NodeConfig part below.
set -euxo pipefail
exec > >(tee /var/log/xrn-boothook.log) 2>&1
date

CLUSTER_NAME=${CLUSTER_NAME}
CLUSTER_REGION=${CLUSTER_REGION}
SATELLITE_ROLE_ARN=${SATELLITE_ROLE_ARN}
# Optional sts:ExternalId. Leave the substitution empty if the cluster-account role's trust
# policy does not require one; if it does, set it to the same value passed to
# `xrnctl setup-iam --external-id`.
EXTERNAL_ID="${EXTERNAL_ID}"
XRN_URL="${XRN_INSTALL_URL}"

# 1. Download xrn-install.
curl -sLfo /usr/local/bin/xrn-install "$XRN_URL"
chmod +x /usr/local/bin/xrn-install

# 2. Kubelet drop-in: run the patch as ExecStartPre. kubelet.service is ordered
#    After=nodeadm-config.service, so the config files exist by the time this runs, and
#    kubelet's main process will not start until the patch succeeds.
EXTERNAL_ID_ARG=""
if [ -n "$EXTERNAL_ID" ]; then
  EXTERNAL_ID_ARG=" --cluster-account-external-id $EXTERNAL_ID"
fi
mkdir -p /etc/systemd/system/kubelet.service.d
cat > /etc/systemd/system/kubelet.service.d/99-xrn-patch.conf <<DROP
[Service]
ExecStartPre=/usr/local/bin/xrn-install patch --cluster-name ${CLUSTER_NAME} --cluster-region ${CLUSTER_REGION} --cluster-account-role-arn ${SATELLITE_ROLE_ARN}${EXTERNAL_ID_ARG}
DROP

systemctl daemon-reload
date
echo "xrn boothook complete"

--//
Content-Type: application/node.eks.aws

---
apiVersion: node.eks.aws/v1alpha1
kind: NodeConfig
spec:
  cluster:
    name: ${CLUSTER_NAME}
    region: ${CLUSTER_REGION}
    apiServerEndpoint: ${CLUSTER_ENDPOINT}
    certificateAuthority: ${CLUSTER_CA}
    cidr: ${CLUSTER_SERVICE_CIDR}
  kubelet:
    flags:
      - --node-labels=eks.amazonaws.com/compute-type=hybrid,xrn.amazonaws.com/satellite-account=${SATELLITE_ACCOUNT_ID}

--//--
