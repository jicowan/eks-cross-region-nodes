#!/bin/bash
# Same-account / cross-region satellite bootstrap (reboot-durable, pre-kubelet ordering).
#
# We patch from a kubelet ExecStartPre drop-in rather than a run-once user-data shell script
# because nodeadm-config.service is WantedBy=multi-user.target and re-runs on EVERY boot,
# regenerating /etc/eks/kubelet/environment and /var/lib/kubelet/kubeconfig from the NodeConfig
# below. A run-once script (text/x-shellscript) executes only on first boot, so after any reboot
# (maintenance, stop/start, crash) nodeadm's regenerated kubeconfig would carry the node's LOCAL
# region again and kubelet could no longer authenticate to the home-region cluster. ExecStartPre
# runs on every kubelet start, ordered After=nodeadm-config.service, so the patch is re-applied
# after each regeneration and survives reboots.
#
# cloud-boothook itself also runs on every boot; here it just (re)installs the binary and the
# drop-in, which is idempotent.
set -euxo pipefail
exec > >(tee /var/log/xrn-boothook.log) 2>&1
date

CLUSTER_NAME=${CLUSTER_NAME}
CLUSTER_REGION=${CLUSTER_REGION}
# /latest/download/ redirects to the most recent release.
XRN_URL="${XRN_INSTALL_URL}"

# 1. Download xrn-install.
curl -sLfo /usr/local/bin/xrn-install "$XRN_URL"
chmod +x /usr/local/bin/xrn-install

# 2. Kubelet drop-in: run the patch as ExecStartPre. kubelet.service is ordered
#    After=nodeadm-config.service, so the config files exist by the time this runs, and
#    kubelet's main process will not start until the patch succeeds. `patch` sets
#    cloud-provider, hostname-override, providerID, topology labels, and rewrites the
#    kubeconfig get-token --region to the cluster's home region. No AssumeRole helper is
#    installed in the same-account case (the instance role can reach the cluster's EKS API).
#
#    --provider-id-format aws writes the standard aws:///<az>/<instance-id> providerID instead
#    of eks-hybrid:///. This is REQUIRED for Cluster Autoscaler to manage these nodes: CA's AWS
#    provider matches ASG instances to Nodes by exact providerID string and only understands the
#    aws:/// form; with eks-hybrid:/// it classifies the nodes as longUnregistered and deletes
#    them on a ~15min loop. Validated 2026-06-24: with --cloud-provider="" the EKS CCM does NOT
#    reap a node carrying the aws:/// providerID, and CA scales the ASG up/down correctly.
#    (Cross-account nodes still use eks-hybrid:/// — see userdata-cross-account.template.txt.)
mkdir -p /etc/systemd/system/kubelet.service.d
cat > /etc/systemd/system/kubelet.service.d/99-xrn-patch.conf <<DROP
[Service]
ExecStartPre=/usr/local/bin/xrn-install patch --cluster-name ${CLUSTER_NAME} --cluster-region ${CLUSTER_REGION} --provider-id-format aws
DROP

systemctl daemon-reload
date
echo "xrn boothook complete"

--//
Content-Type: application/node.eks.aws

---
apiVersion: node.eks.aws/v1alpha1
kind: NodeConfig
spec:
  cluster:
    name: ${CLUSTER_NAME}
    region: ${CLUSTER_REGION}
    apiServerEndpoint: ${CLUSTER_ENDPOINT}
    certificateAuthority: ${CLUSTER_CA}
    cidr: ${CLUSTER_SERVICE_CIDR}
  kubelet:
    flags:
      - --node-labels=eks.amazonaws.com/compute-type=cross-region

--//--
